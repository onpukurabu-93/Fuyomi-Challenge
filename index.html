// 生徒データを保存する配列（すでに定義されている場合は追加不要です） let studentsList = JSON.parse(localStorage.getItem('studentsList')) || []; // 1. 先生が生徒を登録する処理 function registerStudentByTeacher() { const nameInput = document.getElementById('studentName'); const categoryInput = document.getElementById('studentCategory'); const name = nameInput.value.trim(); const category = categoryInput.value; if (!name) { alert('生徒の名前を入力してください。'); return; } // 重複しない4桁のPINコードを自動生成 (例: "1234") let newPin = ""; do { newPin = Math.floor(1000 + Math.random() * 9000).toString(); } while (studentsList.some(s => s.pin === newPin)); // 生徒データオブジェクトを作成 const newStudent = { id: Date.now().toString(), name: name, category: category, pin: newPin }; studentsList.push(newStudent); localStorage.setItem('studentsList', JSON.stringify(studentsList)); alert(`【登録完了】\n名前: ${name}\n区分: ${category}\n生徒に伝えるPINコード: ${newPin}`); // フォームのリセット nameInput.value = ''; } // 2. 生徒が初回PINコードを入力してログインする処理 function loginWithPin() { const pinValue = document.getElementById('pinInput').value.trim(); if (!pinValue) { alert('コードを入力してください。'); return; } // 入力されたPINコードと一致する生徒を探す const matchedStudent = studentsList.find(s => s.pin === pinValue); if (matchedStudent) { // ログイン成功：ブラウザにログイン状態（生徒情報）を保存 localStorage.setItem('currentStudent', JSON.stringify(matchedStudent)); alert(`${matchedStudent.name} さん、ようこそ！次回からは自動でログインされます。`); // 入力欄をクリア document.getElementById('pinInput').value = ''; // TODO: ここでメニュー画面へ遷移・表示切り替えを行います } else { alert('コードが違います。先生に確認してください。'); } } // 3. アプリ起動時に「すでにログイン済みか」を確認する処理 function checkAutoLogin() { const savedStudent = localStorage.getItem('currentStudent'); if (savedStudent) { const currentStudent = JSON.parse(savedStudent); // すでにログインしている場合の処理（例：コード入力画面を隠してメニューを表示） console.log("ログイン中の生徒:", currentStudent.name); } } // 画面読み込み時に自動ログインをチェック window.addEventListener('DOMContentLoaded', checkAutoLogin);
